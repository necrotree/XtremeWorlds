    Private Sub picWebsite_Click()
        Dim address As String
        address = Trim$(WEBSITE)
        If Len(address) = 0 Then
            MsgBox "The server has not supplied a website address yet.", vbInformation, GAME_NAME
            Exit Sub
        End If
        If LCase$(Left$(address, 7)) <> "http://" And LCase$(Left$(address, 8)) <> "https://" Then address = "https://" & address
        ShellExecute Me.hWnd, "open", address, vbNullString, vbNullString, 1
    End Sub

    Public Sub SetCharacterSlot(ByVal slot As Long, ByVal name As String, ByVal sprite As Long)
        Dim preview As Object, label As Object
        Dim source As RECT, destination As RECT
        If slot < 0 Or slot > 2 Then Exit Sub
        Set preview = Me.Controls("picCharacter" & slot)
        Set label = Me.Controls("lblCharacter" & slot)
        label.Caption = name
        If Len(name) = 0 Then label.Caption = "Empty"
        label.ToolTipText = label.Caption
        mCharacterSprites(slot) = sprite
        preview.Cls
        If Len(name) = 0 Or DD_SpriteSurf Is Nothing Then Exit Sub
        If sprite < 0 Or (sprite + 1) * PIC_Y > DD_SpriteSurf.Height Then Exit Sub
        source.Left = 3 * PIC_X
        source.Top = sprite * PIC_Y
        source.Right = source.Left + PIC_X
        source.Bottom = source.Top + PIC_Y
        destination.Left = 8: destination.Top = 8
        destination.Right = 40: destination.Bottom = 40
        DD_SpriteSurf.BltToDC preview.hDC, source, destination
        preview.Refresh
    End Sub

    Public Sub SelectCharacterSlot(ByVal slot As Long)
        Dim i As Long, label As Object
        If slot < 0 Or slot >= lstChars.ListCount Then Exit Sub
        lstChars.ListIndex = slot
        For i = 0 To 2
            Set label = Me.Controls("lblCharacter" & i)
            label.ForeColor = RGB(191, 205, 128)
            label.FontBold = (i = slot)
            If i = slot Then label.ForeColor = vbWhite
        Next
    End Sub
